03 · Tecnología

¿Qué es una soldadora de punto?

Une piezas metálicas mediante el calor generado por corriente eléctrica en una zona pequeña. Los electrodos aplican presión y permiten que la corriente atraviese el material, formando un punto de unión sólido.

¿Por qué se produce el calor?

La corriente eléctrica atraviesa una resistencia en la zona de contacto y convierte la energía eléctrica en energía térmica, concentrada en un punto pequeño que permite la unión metálica.

El brazo extendido es el caso más exigente.

Con el brazo a 0° (horizontal), el torque en la articulación es τ = m · g · d. Con 1,5 kg a 0,85 m da 12,51 Nm nominales; con factor de seguridad 2, el actuador debe poder entregar 25,02 Nm. Lo vemos en una diapositiva de un brazo de referencia y en esta calculadora.

Cómo funciona el control

En hombro, codo y muñeca el Arduino compara el ángulo que lee el encoder con el objetivo y corrige la posición, con un máximo de 200 pasos por ciclo. La base no tiene encoder, así que se mueve por pasos, sin retroalimentación.

Prueba de dos servos prueba_de_servos.txt
#include <Servo.h>

Servo Servo1;
Servo Servo2;

byte angulo1 = 0;
byte angulo2 = 0;

byte estado = 1;


void setup() {

  Servo1.attach(9);
  Servo2.attach(11);

  Servo1.write(angulo1);
  Servo2.write(angulo2);
}


void loop() {

  // =========================================
  // ESTADO 1
  // Servo 1: 0 -> 80
  // Servo 2: espera en 0
  // =========================================

  if (estado == 1)
  {
    angulo1++;

    Servo1.write(angulo1);

    if (angulo1 >= 80)
    {
      estado = 2;
    }
  }


  // =========================================
  // ESTADO 2
  // Servo 1: 80 -> 160
  // Servo 2: 0 -> 40
  //
  // Servo 2 se mueve a la mitad
  // de velocidad que Servo 1
  // =========================================

  if (estado == 2)
  {
    angulo1++;

    if (angulo1 % 2 == 0)
    {
      angulo2++;
    }

    Servo1.write(angulo1);
    Servo2.write(angulo2);

    if (angulo1 >= 160)
    {
      estado = 3;
    }
  }


  // =========================================
  // ESTADO 3
  // Servo 1: 160 -> 80
  // Servo 2: 40 -> 0
  //
  // Servo 2 vuelve a la mitad
  // de velocidad
  // =========================================

  if (estado == 3)
  {
    angulo1--;

    if (angulo1 % 2 == 0)
    {
      angulo2--;
    }

    Servo1.write(angulo1);
    Servo2.write(angulo2);

    if (angulo1 <= 80)
    {
      estado = 4;
    }
  }


  // =========================================
  // ESTADO 4
  // Servo 1: 80 -> 0
  // Servo 2: espera en 0
  // =========================================

  if (estado == 4)
  {
    angulo1--;

    Servo1.write(angulo1);

    if (angulo1 <= 0)
    {
      estado = 1;
    }
  }


  // Controla la velocidad del movimiento
  delay(20);
}

#include <Servo.h>

Servo servo1;

byte angulo1, angulo2, angulo3, angulo4;
int automatico = 9;
int grabando = 7;
int para de grabar = 8;
int reset = 13;
byte movimientos[1000][4];
int cantidad = 0;
bool grabando = false;
bool reproduciendo = false;

void setup() {
  servo1.attach(6);
  servo2.attach(2);
  servo3.attach(3);
  servo4.attach(4);
  servo5.attach(5);

  pinMode(grabando, INPUT);
  pinMode(para de grabar, INPUT);
  pinMode(automatico, INPUT);
  pinMode(reset, INPUT);
}

void loop() {
  if (digitalRead(grabando) == HIGH) {
    grabando = true;
    reproduciendo = false;
    delay(300);
  }

  if (digitalRead(para de grabar) == HIGH) {
    grabando = false;
    reproduciendo = false;
    delay(300);
  }

  if (digitalRead(automatico) == HIGH) {
    grabando = false;
    reproduciendo = true;
    delay(300);
  }

  if (digitalRead(reset) == HIGH) {
    reproduciendo = false;
    grabando = false;
    delay(300);
  }

  if (grabando) {
    angulo1 = map(analogRead(A0), 0, 1023, 0, 90);
    angulo2 = map(analogRead(A1), 0, 1023, 0, 90);
    angulo3 = map(analogRead(A2), 0, 1023, 0, 90);
    angulo4 = map(analogRead(A3), 0, 1023, 0, 90);

    servo1.write(angulo1);
    servo2.write(angulo2);
    servo3.write(angulo3);
    servo4.write(angulo4);

    if (cantidad < 1000) {
      movimientos[cantidad][0] = angulo1;
      movimientos[cantidad][1] = angulo2;
      movimientos[cantidad][2] = angulo3;
      movimientos[cantidad][3] = angulo4;
      cantidad++;
    }
    delay(20);
  }

  if (reproduciendo) {
    for (int i = 0; i < cantidad; i++) {
      if (digitalRead(reset) == HIGH) {
        reproduciendo = false;
        delay(300);
        break;
      }

      servo1.write(movimientos[i][0]);
      servo2.write(movimientos[i][1]);
      servo3.write(movimientos[i][2]);
      servo4.write(movimientos[i][3]);

      delay(20);
    }
  }

  if (!grabando && !reproduciendo) {
    angulo1 = map(analogRead(A0), 0, 1023, 0, 90);
    angulo2 = map(analogRead(A1), 0, 1023, 0, 90);
    angulo3 = map(analogRead(A2), 0, 1023, 0, 90);
    angulo4 = map(analogRead(A3), 0, 1023, 0, 90);

    servo1.write(angulo1);
    servo2.write(angulo2);
    servo3.write(angulo3);
    servo4.write(angulo4);
  }
}

// =====================================================
//             SELECCIONAR TCA9548A
// =====================================================
void seleccionarCanal(byte canal)
{
  Wire.beginTransmission(TCA9548A);
  Wire.write(1 << canal);
  Wire.endTransmission();
}
// =====================================================
//                 LEER AS5600
// =====================================================
float leerAS5600(byte canal)
{
  seleccionarCanal(canal);
  Wire.beginTransmission(AS5600);
  Wire.write(AS5600_ANGLE_MSB);
  Wire.endTransmission(false);
  Wire.requestFrom(AS5600, 2);
  if (Wire.available() < 2)
  {
    return 0;
  }
  byte msb = Wire.read();
  byte lsb = Wire.read();
  int valor =
    ((msb << 8) | lsb) & 0x0FFF;
  float angulo =
    valor * 360.0 / 4096.0;
  return angulo;
}
// =====================================================
//            OBTENER POSICIÓN REAL
// =====================================================
float posicionEncoder(
  byte canal,
  float offset,
  bool invertir
)
{
  float angulo =
    leerAS5600(canal);
  angulo -= offset;
  if (invertir)
  {
    angulo = -angulo;
  }
  while (angulo < 0)
  {
    angulo += 360;
  }
  while (angulo >= 360)
  {
    angulo -= 360;
  }
  return angulo;
}
// =====================================================
//                 CONTROL HOMBRO
// =====================================================
void controlarHombro()
{
  float actual =
    posicionEncoder(
      CANAL_HOMBRO,
      offsetHombro,
      invertirHombro
    );
  float error =
    objetivoHombro - actual;
  // -------------------------------
  // Ya está en posición
  // -------------------------------
  if (abs(error) <= tolerancia)
  {
    return;
  }
  // -------------------------------
  // Calcular pasos necesarios
  // -------------------------------
  long pasos =
    abs(error) * pasosHombroGrado;
  if (pasos < 1)
  {
    pasos = 1;
  }
  if (pasos > maxCorreccion)
  {
    pasos = maxCorreccion;
  }
  // -------------------------------
  // Corregir posición
  // -------------------------------
  if (error > 0)
  {
    motorHombro.move(pasos);
  }
  else
  {
    motorHombro.move(-pasos);
  }
}

      1023,
      -180,
      180
    );
  // HOMBRO
  objetivoHombro =
    map(
      analogRead(POT_HOMBRO),
      0,
      1023,
      0,
      90
    );
  // CODO
  objetivoCodo =
    map(
      analogRead(POT_CODO),
      0,
      1023,
      0,
      90
    );
  // MUÑECA
  objetivoMuneca =
    map(
      analogRead(POT_MUNECA),
      0,
      1023,
      0,
      90
    );
}
// =====================================================
//                    CONTROL BASE
// =====================================================
void controlarBase()
{
  // Convertir grados a pasos
  long pasosObjetivo =
    objetivoBase *
    pasosBaseVuelta /
    360.0;
  posicionBase =
    pasosObjetivo;
  motorBase.moveTo(
    posicionBase
  );
}
// =====================================================
//                GUARDAR MOVIMIENTO
// =====================================================
void guardarMovimiento()
{
  archivo =
    SD.open(
      "MOV.TXT",
      FILE_WRITE
    );
  if (!archivo)
  {
    Serial.println("Error abriendo MOV.TXT");
    return;
  }
  // Guardamos LOS OBJETIVOS.
  //
  // En los motores con encoder:
  // el objetivo representa el ángulo
  // que queremos que tenga la articulación.
  archivo.print(objetivoBase, 2);
  archivo.print(",");
  archivo.print(objetivoHombro, 2);
  archivo.print(",");
  archivo.print(objetivoCodo, 2);
  archivo.print(",");
  archivo.println(objetivoMuneca, 2);
  archivo.close();
}
// =====================================================
//              REPRODUCIR MOVIMIENTOS
// =====================================================
void reproducirMovimiento()
{
  archivo =
    SD.open("MOV.TXT");
  if (!archivo)
  {
    Serial.println("No existe MOV.TXT");
    reproduciendo = false;
    return;
  }
  while (archivo.available())
  {
    // ---------------------------------------------
    // BOTÓN PARAR
    // ---------------------------------------------
    if (digitalRead(BOTON_PARAR) == HIGH)
    {
      reproduciendo = false;
      break;
    }
    // ---------------------------------------------
    // LEER UNA LÍNEA
    // ---------------------------------------------
    String linea =
      archivo.readStringUntil('\n');
    linea.trim();
    if (linea.length() == 0)
    {
      continue;
    }
    // ---------------------------------------------
    // SEPARAR LOS 4 VALORES
    // ---------------------------------------------
    int c1 = linea.indexOf(',');
    int c2 =
      linea.indexOf(
        ',',
        c1 + 1
      );
    int c3 =
      linea.indexOf(
        ',',
        c2 + 1
      );
    if (
      c1 < 0 ||
      c2 < 0 ||
      c3 < 0
    )
    {
      continue;
    }
    objetivoBase =
      linea.substring(
        0,
        c1
      ).toFloat();
    objetivoHombro =
      linea.substring(
        c1 + 1,
        c2
      ).toFloat();
    objetivoCodo =
      linea.substring(
        c2 + 1,
        c3
      ).toFloat();
    objetivoMuneca =
      linea.substring(
        c3 + 1
      ).toFloat();
    // ---------------------------------------------
    // BASE
    // ---------------------------------------------
    controlarBase();
    // ---------------------------------------------
    // ESPERAR HASTA LLEGAR
    // ---------------------------------------------
    bool terminado = false;
    while (!terminado)
    {
      ejecutarMotores();
      // Controlar las tres articulaciones
      // con encoder.
      if (
        millis() - ultimoControl
        >= intervaloControl
      )
      {
        ultimoControl = millis();
        controlarHombro();
        controlarCodo();
        controlarMuneca();
      }
      // -------------------------------------------
      // PARAR
      // -------------------------------------------
      if (
        digitalRead(BOTON_PARAR)
        == HIGH
      )
      {
        reproduciendo = false;
        motorBase.stop();
        motorHombro.stop();
        motorCodo.stop();
        motorMuneca.stop();
        break;
      }
      // -------------------------------------------
      // COMPROBAR POSICIONES
      // -------------------------------------------
      float hombro =
        posicionEncoder(
          CANAL_HOMBRO,
          offsetHombro,
          invertirHombro
        );
      float codo =
        posicionEncoder(
          CANAL_CODO,
          offsetCodo,
          invertirCodo
        );
      float muneca =
        posicionEncoder(
          CANAL_MUNECA,
          offsetMuneca,
          invertirMuneca
        );
      bool hombroOK =
        abs(
          objetivoHombro - hombro
        ) <= tolerancia;
      bool codoOK =
        abs(
          objetivoCodo - codo
        ) <= tolerancia;
      bool munecaOK =
        abs(
          objetivoMuneca - muneca
        ) <= tolerancia;
      bool baseOK =
        motorBase.distanceToGo() == 0;
      if (
        hombroOK &&
        codoOK &&
        munecaOK &&
        baseOK
      )
      {
        terminado = true;
      }
    }
    if (!reproduciendo)
    {
      break;
    }
    // Tiempo entre movimientos
    delay(20);
  }
  archivo.close();
  reproduciendo = false;
  Serial.println(
    "Reproduccion terminada."
  );
}
// =====================================================
//                       SETUP
// =====================================================
void setup()
{
  Serial.begin(115200);
  Wire.begin();
  // ---------------------------------------------
  // PINES
  // ---------------------------------------------
  pinMode(
    ENABLE_A4988,
    OUTPUT
  );
  pinMode(
    ENABLE_TB6600,
    OUTPUT
  );
  pinMode(
    BOTON_GRABAR,
    INPUT
  );
  pinMode(
    BOTON_AUTO,
    INPUT
  );
  pinMode(
    BOTON_PARAR,
    INPUT
  );
  // ---------------------------------------------
  // HABILITAR DRIVERS
  // ---------------------------------------------
  // A4988: LOW = habilitado
  digitalWrite(
    ENABLE_A4988,
    LOW
  );
  // TB6600:
  // LOW normalmente habilita si está
  // cableado con lógica activa baja.
  digitalWrite(
    ENABLE_TB6600,
    LOW
  );
  // ---------------------------------------------
  // VELOCIDAD BASE
  // ---------------------------------------------
  motorBase.setMaxSpeed(2500);
  motorBase.setAcceleration(1200);
  // ---------------------------------------------
  // HOMBRO
  // ---------------------------------------------
  motorHombro.setMaxSpeed(2500);
  motorHombro.setAcceleration(1000);
  // ---------------------------------------------
  // CODO
  // ---------------------------------------------
  motorCodo.setMaxSpeed(2500);
  motorCodo.setAcceleration(1000);
  // ---------------------------------------------
  // MUÑECA
  // ---------------------------------------------
  motorMuneca.setMaxSpeed(2500);
  motorMuneca.setAcceleration(1000);
  // ---------------------------------------------
  // SD
  // ---------------------------------------------
  if (SD.begin(SD_CS))
  {
    Serial.println(
      "Tarjeta SD OK"
    );
  }
  else
  {
    Serial.println(
      "ERROR: tarjeta SD"
    );
  }
  Serial.println(
    "Brazo robotico iniciado."
  );
  delay(1000);
}
// =====================================================
//                       LOOP
// =====================================================
void loop()
{
  ejecutarMotores();
  // =================================================
  // BOTÓN PARAR
  // =================================================
  if (
    digitalRead(BOTON_PARAR)
    == HIGH
  )
  {
    grabando = false;
    reproduciendo = false;
    motorBase.stop();
    motorHombro.stop();
    motorCodo.stop();
    motorMuneca.stop();
    Serial.println("BRAZO PARADO");
    delay(300);
  }
  // =================================================
  // BOTÓN GRABAR
  // =================================================
  if (
    digitalRead(BOTON_GRABAR)
    == HIGH &&
    !grabando
  )
  {
    grabando = true;
    reproduciendo = false;
    Serial.println("GRABANDO");
    delay(300);
  }
  // =================================================
  // BOTÓN AUTOMÁTICO
  // =================================================
  if (
    digitalRead(BOTON_AUTO)
    == HIGH &&
    !reproduciendo
  )
  {
    grabando = false;
    reproduciendo = true;
    Serial.println("REPRODUCIENDO");
    delay(300);
    reproducirMovimiento();
    return;
  }
  // =================================================
  // MODO MANUAL
  // =================================================
  if (
    !grabando &&
    !reproduciendo
  )
  {
    leerPotenciometros();
    controlarBase();
    if (
      millis() - ultimoControl
      >= intervaloControl
    )
    {
      ultimoControl = millis();
      controlarHombro();
      controlarCodo();
      controlarMuneca();
    }
  }
  // =================================================
  // MODO GRABACIÓN
  // =================================================
  if (grabando)
  {
    leerPotenciometros();
    controlarBase();
    if (
      millis() - ultimoControl
      >= intervaloControl
    )
    {
      ultimoControl = millis();
      controlarHombro();
      controlarCodo();
      controlarMuneca();
      guardarMovimiento();
    }
  }
}
